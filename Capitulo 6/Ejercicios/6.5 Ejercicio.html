Ejercicio 6.5 - Instrucciones por tarea

// a) Este programa calcula el producto de tres enteros

// b)
var x, y, z, resultado;

// c)
var xVal, yVal, zVal;

// d)
xVal = window.prompt( "Introduzca el primer valor:", "0" );

// e)
yVal = window.prompt( "Introduzca el segundo valor:", "0" );

// f)
zVal = window.prompt( "Introduzca el tercer valor:", "0" );

// g)
x = parseInt( xVal );

// h)
y = parseInt( yVal );

// i)
z = parseInt( zVal );

// j)
resultado = x * y * z;

// k)
document.writeln( "<h1>El producto es " + resultado + "</h1>" );
